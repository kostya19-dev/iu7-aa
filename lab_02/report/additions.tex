\begin{lstlisting}[label={lst:recursive}, caption={Рекурсивная функция вывода чисел в диапазоне от А до В}]
	void print_numbers_recursive(int a, int b)
	{
		if (b > a)
		{
			print_numbers_recursive(a, b - 1);
		}
		
		std::cout << b;
	}
\end{lstlisting}

\begin{lstlisting}[label={lst:iterative}, caption={Итеративная функция вывода чисел в диапазоне от А до В}]
	void print_numbers_iterative(int a, int b)
	{
		if (a <= b)
		{
			for (int i = a; i <= b; i++)
			{
				std::cout << i;
				if (i < b)
				{
					std::cout << " ";
				}
			}
		}
	}
\end{lstlisting}
