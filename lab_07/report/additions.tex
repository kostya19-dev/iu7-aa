\begin{lstlisting}[language=Python,label={lst:chatgpt_core},
caption={Извлечение строк PDF и формирование базовых шаблонов (модель ChatGPT)}]
from __future__ import annotations

import re
from typing import List, Tuple
from PyPDF2 import PdfReader

Line = Tuple[str, int, int]

def spaced_letters(word: str) -> str:
    return r"\s*".join(re.escape(ch) for ch in word)

W_TOC_1 = "SODERZHANIE"
W_TOC_2 = "OGLAVLENIE"
W_INTRO = "VVEDENIE"
W_CONCL = "ZAKLYUCHENIE"
W_LIST  = "SPISOK"
W_USED  = "ISPOLSOVANNYH"
W_SRC   = "ISTOCHNIKOV"

RX_TOC = re.compile(rf"^\s*(?:{W_TOC_1}|{W_TOC_2})\s*$", re.IGNORECASE)
RX_INTRO = re.compile(rf"^\s*{spaced_letters(W_INTRO)}\b", re.IGNORECASE)
RX_CONCL = re.compile(rf"^\s*{spaced_letters(W_CONCL)}\b", re.IGNORECASE)
RX_REFS = re.compile(
    rf"^\s*{spaced_letters(W_LIST)}\s+"
    rf"{spaced_letters(W_USED)}\s+"
    rf"{spaced_letters(W_SRC)}\b",
    re.IGNORECASE
)

def extract_lines(pdf_path: str) -> List[Line]:
    reader = PdfReader(pdf_path)
    result: List[Line] = []

    for page_no, page in enumerate(reader.pages, start=1):
        text = page.extract_text() or ""
        for line_no, raw in enumerate(text.splitlines(), start=1):
            line = raw.strip()
            if line:
                result.append((line, page_no, line_no))

    return result

\begin{lstlisting}[language=Python,label={lst:deepseek_core},
caption={Определение границ оглавления и проверка структуры (модель DeepSeek)}]
import re
from typing import List, Optional, Tuple

Line = Tuple[str, int, int]
Match = Tuple[str, Tuple[int, int]]

RX_TOC_ENTRY = re.compile(r"^(?=.*[A-Za-z])(?=.*\d).+\d+$")

def toc_bounds(lines: List[Line], limit: int = 500) -> Tuple[int, int]:
    start = next(
        (i for i, (text, _, _) in enumerate(lines) if RX_TOC.match(text)),
        -1
    )
    if start < 0:
        return -1, -1

    end = start + 1
    miss = 0

    for i in range(start + 1, min(len(lines), start + limit)):
        if RX_TOC_ENTRY.match(lines[i][0]):
            end = i + 1
            miss = 0
        else:
            miss += 1
            if miss >= 5:
                break

    return start, end

def first_match(where: List[Line], rx: re.Pattern) -> Optional[Line]:
    for text, page, line in where:
        if rx.search(text):
            return (text, page, line)
    return None

def check_structure(lines: List[Line]) -> Tuple[bool, List[Match]]:
    a, b = toc_bounds(lines)
    if a < 0:
        return False, []

    toc = lines[a:b]
    body = lines[:a] + lines[b:]

    intro = first_match(toc, RX_INTRO)
    concl = first_match(toc, RX_CONCL)
    refs  = first_match(toc, RX_REFS)

    if not (intro and concl and refs):
        return False, []

    def heading_rx(rx: re.Pattern) -> re.Pattern:
        return re.compile(
            rf"^\s*(?:\d+(?:\.\d+)*)?\s*({rx.pattern})\s*$",
            re.IGNORECASE
        )

    if (first_match(body, heading_rx(RX_INTRO)) or
        first_match(body, heading_rx(RX_CONCL)) or
        first_match(body, heading_rx(RX_REFS))):
        return False, []

    return True, [
        (intro[0], (intro[1], intro[2])),
        (concl[0], (concl[1], concl[2])),
        (refs[0],  (refs[1],  refs[2])),
    ]

\begin{lstlisting}[language=Python,label={lst:qwen_core},
caption={Оптимизация досрочного завершения проверки (модель Qwen)}]
from typing import Dict, List, Optional, Tuple

Line = Tuple[str, int, int]
Match = Tuple[str, Tuple[int, int]]

def find_required_in_toc(toc: List[Line]) -> Optional[List[Match]]:
    found: Dict[str, Optional[Match]] = {
        "intro": None,
        "concl": None,
        "refs": None
    }

    for text, page, line in toc:
        if found["intro"] is None and RX_INTRO.search(text):
            found["intro"] = (text, (page, line))
        elif found["concl"] is None and RX_CONCL.search(text):
            found["concl"] = (text, (page, line))
        elif found["refs"] is None and RX_REFS.search(text):
            found["refs"] = (text, (page, line))

        if all(found.values()):
            return [
                found["intro"],
                found["concl"],
                found["refs"]
            ]

    return None
